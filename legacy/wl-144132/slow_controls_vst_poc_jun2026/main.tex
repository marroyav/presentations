\begin{frame}[t]{Development Gaps Before Production}
\eyebrow{WHAT THE VST PROOF DOES NOT YET DEFINE}
\vspace{1.4mm}

\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{beamercolorbox}[wd=\linewidth,sep=5pt,rounded=true]{panel alt}
{\small\bfseries\textcolor{noircyan}{Interfaces to freeze}\par}
\vspace{1mm}
\gapline{Hardware database}{
SC schema and versioned JSON exports for device, endpoint, rack/slot, IP, outlet, owner, and dependency edges.
}
\gapline{OPC-UA namespaces}{
Generated NodeIds, units, limits, quality, timestamps, ownership, and alarm metadata; no hand-maintained map.
}
\gapline{Missing contracts}{
ZMQ/protobuf schemas, command acknowledgements, error taxonomy, retry rules, lifecycle states, and health readbacks.
}
\end{beamercolorbox}

\column{0.49\textwidth}
\begin{beamercolorbox}[wd=\linewidth,sep=5pt,rounded=true]{panel alt}
{\small\bfseries\textcolor{noircyan}{Control path still open}\par}
\vspace{1mm}
\gapline{Interlocks and sensors}{
PDU/PLC state, smoke sensors, trip latches, bounds, write-enable logic, and fail-closed behavior.
}
\gapline{Migration layers}{
C++ reference path, Rust candidate path, and RPU-owned AFE access need shadow-mode tests before writes.
}
\gapline{Integration holes}{
Ignition import, historian policy, alarm classes, deployment, credentials, subnets, and power-cycle recovery.
}
\end{beamercolorbox}
\end{columns}
\end{frame}

\begin{frame}[t]{Rust vs C++: Decisions Still Open}
\eyebrow{C++ IS THE LIVE HARDWARE REFERENCE; RUST IS THE COMPATIBILITY AND RPU-SAFETY PATH}
\vspace{1mm}

\begin{columns}[T,onlytextwidth]
\column{0.32\textwidth}
\begin{exampleblock}{Tested reference}
\scriptsize
\begin{itemize}\tightitem
  \item Legacy C++ \texttt{daphneServer} replies on DAPHNE-15 \texttt{40001}.
  \item \texttt{ControlEnvelopeV2} readbacks are the current compatibility contract.
  \item C++ OPC-UA bridge reads live values on ONL.
\end{itemize}
\end{exampleblock}

\column{0.32\textwidth}
\begin{block}{Rust target}
\scriptsize
\begin{itemize}\tightitem
  \item Protocol-compatible server on a separate endpoint until parity.
  \item Linux aggregates status and enforces preflight.
  \item AFE-affecting commands fail closed without a valid RPU backend.
\end{itemize}
\end{block}

\column{0.32\textwidth}
\begin{alertblock}{Open questions}
\scriptsize
\begin{itemize}\tightitem
  \item What response parity lets Rust replace or front C++?
  \item Does OPC-UA remain a small C++ bridge or move into Rust?
  \item What RPU ABI test is required before any AFE write migration?
\end{itemize}
\end{alertblock}
\end{columns}

\vspace{2mm}
{\footnotesize
Boundary: C++ legacy service is the tested hardware reference; Rust is for protocol-compatible service evolution and RPU-owned safety.
}
\end{frame}
