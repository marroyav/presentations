\begin{frame}[t]{What must be captured in the matrix?}
\eyebrow{ENOUGH INFORMATION TO TURN EACH INTERACTION INTO WORK AND EVIDENCE}
\vspace{1.5mm}

\begin{columns}[T,onlytextwidth]
\column{0.32\textwidth}
\fixedpanel{3.7cm}{Identity and authority}{
\begin{itemize}\tightitem
  \item Source and destination.
  \item Variable or command; type, unit, range, rate.
  \item Command owner and permitted actor.
  \item Applicable lifecycle stage.
  \item Source ICD/schema and controlled revision.
\end{itemize}
}

\column{0.32\textwidth}
\fixedpanel{3.7cm}{Constraints and response}{
\begin{itemize}\tightitem
  \item Preconditions and allowed detector states.
  \item Permits, inhibits, and interlocks.
