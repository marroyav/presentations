\begin{frame}{Three situations need three different responses}
  \DuneColorBlock{coral}{PROTECT EQUIPMENT}{LAr level is not confirmed above the minimum}
    {HV is unavailable: voltage-setting, enabling and ramping are blocked.}
  \DuneColorBlock{paper}{TAKE TURNS}{Camera lights or an external laser with PDS}
    {Block the second request. Confirm the light is off before PDS returns.}
  \DuneColorBlock{sky}{KEEP RUNNING; LABEL DATA}{Purity monitor with PDS}
    {Keep the photon detection system (PDS) running. Mark the affected data.}
\end{frame}

\begin{frame}{HV stays unavailable until LAr is above the minimum}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \DuneFixedColorBlock{coral}{3.3cm}{HV UNAVAILABLE}
        {LAr level is too low or is not reliably known}
        {Voltage-setting, enabling and ramping commands remain blocked.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \DuneFixedColorBlock{paper}{3.3cm}{LEVEL REQUIREMENT MET}
        {Measured LAr level is above the required minimum}
        {All required instrumentation is covered, including the cathode. Other HV checks must also pass.}
    \end{column}
  \end{columns}
  \medskip
  \DuneTakeaway{This is a prerequisite for HV availability. Reaching the level does not switch HV on.}
  \RuleNote{P-001 · Cryogenics + HV + Detector Protection specify the minimum level, reference point and measurement checks.}
\end{frame}

    \end{column}

      \DuneFixedColorBlock{sky}{4.2cm}{A PURITY-ACTIVITY RECORD IS MISSING}
