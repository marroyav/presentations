\begin{frame}{Every rule answers two questions}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \DuneFixedColorBlock{paper}{4.40cm}{1 · WHAT IS AT STAKE?}
        {Consequence class}
        {\textbf{PS / ED}\quad Personnel, facility, or equipment protection\par\medskip
         \textbf{OP}\quad Operating conflict or loss of availability\par\medskip
         \textbf{DQ}\quad Physics or data-quality contamination\par\medskip
         \textbf{OK / UNK}\quad Compatible, or not yet classified}
    \end{column}
    \begin{column}{0.48\textwidth}
      \DuneFixedColorBlock{sky}{4.40cm}{2 · HOW IS IT HANDLED?}
        {Enforcement class}
        {\textbf{FSS / AUTO}\quad Independent safe state, trip, or inhibit\par\medskip
         \textbf{SW / HOLD}\quad Software denial or named authorization\par\medskip
         \textbf{TAG / NONE}\quad Allow, record, or take no special action\par\medskip
         \textbf{TBD}\quad Not creditable until owners resolve it}
    \end{column}
  \end{columns}

  \smallskip
  \DuneTakeaway{A cell such as \RuleID{ED/AUTO [P-002]} states both the reason and the response.}
\end{frame}
