\newcommand{\ScanGrid}[2]{%
  \begin{tikzpicture}[x=0.66cm,y=0.66cm]
    \fill[DunePaper] (0,0) rectangle (3,3);
    \draw[step=0.66cm,DuneBorder,line width=0.5pt] (0,0) grid (3,3);
    \foreach \x in {0.5,1.5,2.5}
      \foreach \y in {0.5,1.5,2.5}
        \fill[DuneOnAccent] (\x,\y) circle (2pt);
    \draw[-{Stealth},DuneSky,line width=1pt] (-0.12,-0.12) -- (3.5,-0.12);
    \draw[-{Stealth},DuneSky,line width=1pt] (-0.12,-0.12) -- (-0.12,3.5);
    \node[anchor=north,text=DuneText,font=\small] at (1.5,-0.38) {#1};
    \node[rotate=90,anchor=south,text=DuneText,font=\small] at (-0.38,1.5) {#2};
  \end{tikzpicture}%
}
