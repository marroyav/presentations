\begin{frame}{Why \(14\times512\times32\) is not yet throughput}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \DuneFixedColorBlock{paper}{2.65cm}{BITS IN ONE BOARD-WIDE SET}{A size, not a rate}
        {\(14\,\mathrm{bit/sample}\times512\,\mathrm{sample/frame}\times32\,\mathrm{ch}=229{,}376\,\mathrm{bit}\).}
    \end{column}
    \begin{column}{0.49\textwidth}
      \DuneFixedColorBlock{sky}{2.65cm}{BITS PER SECOND}{Multiply by activity}
        {With the modeled record overhead: \(32\,\mathrm{ch}\times R_{\rm ch}\times7{,}680\,\mathrm{bit/frame}\).}
    \end{column}
  \end{columns}

  \medskip
  \DuneCompactColorBlock{coral}{512 + 50\% OVERLAP}{Overlap changes acceptance opportunities, not record size}
    {Each fragment is still 7,680 bits. The 256 overlapping samples are retransmitted unless firmware coalesces them: \(T=R_{\rm accepted,overlap}\times7{,}680\).}

\end{frame}

\begin{frame}{Hermes back-pressure changes acceptance}
  \framesubtitle{Intrinsic-LAr + one-window gamma pilot; homogeneous-board illustration}
  \footnotesize
  \textbf{Sender input:} $T_{\rm in}=R_{\rm accepted,local}B_{\rm record}$;
  queue growth: $dQ/dt=\max(0,T_{\rm in}-C_{\rm payload})$, where
  $C_{\rm payload}<10$ Gbit/s.

  \centering
  \setlength{\tabcolsep}{4pt}
  \renewcommand{\arraystretch}{1.00}
  \begin{tabular}{@{}lrrl@{}}
    \toprule
    Population & 512 demand $\rightarrow T_{\rm in}$ & 1024 demand $\rightarrow T_{\rm in}$ & Ideal-link test \\
    \midrule
    HD & 13.47 $\rightarrow$ 8.03 & 26.05 $\rightarrow$ \textbf{12.49} & 1024 saturates \\
    VD cathode & 22.70 $\rightarrow$ \textbf{11.87} & 43.90 $\rightarrow$ \textbf{16.41} & \textbf{both saturate} \\
    VD long wall & 13.09 $\rightarrow$ 7.91 & 25.31 $\rightarrow$ \textbf{11.89} & 1024 saturates \\
    VD short wall & 9.38 $\rightarrow$ 6.45 & 18.14 $\rightarrow$ \textbf{10.10} & 1024 at limit \\
    \bottomrule
  \end{tabular}

  All traffic values are Gbit/s for one homogeneous board and one 10-Gbit/s sender. ``Demand'' is before the local gate; $T_{\rm in}$ is after it.

  \DuneCompactColorBlock{coral}{CONSEQUENCE}{The ideal 10-Gbit/s comparison is already unfavorable}
    {All 1024-sample populations meet or exceed line rate; VD cathode also exceeds it at 512. Usable payload is below 10 Gbit/s, so final loss requires Hermes ready, FIFO depth, and the real channel-to-sender map.}
\end{frame}
