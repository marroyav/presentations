\begin{frame}{The Whole Proposal}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.50\textwidth}
    \DuneEyebrow{ONE QUESTION}
    {\color{white}\bfseries\fontsize{16}{19}\selectfont
    Who owns the software logic that represents detector hardware to operators, DAQ, Computing, and protection systems?\par}
    \medskip
    \begin{itemize}
      \DuneTightList
      \item Start with ownership, then choose repository names.
      \item Keep device-control logic close to Slow Controls when SC owns the bridge.
      \item Keep DAQ and Computing interfaces explicit and testable.
    \end{itemize}
  \end{column}
  \begin{column}{0.46\textwidth}
    \DuneFixedCard[dune accent]{1.20cm}{Slow Controls}
      {Owns bridge logic, OPC-UA representation, operator alarms, history, and SC-facing command semantics.}
      {}
    \smallskip
    \DuneFixedCard[dune evidence]{1.20cm}{DAQ and Computing}
      {DAQ owns DAQ clients and run-control behavior. Computing owns platform support, CI, artifacts, and deployment infrastructure.}
      {}
    \smallskip
    \DuneFixedCard[dune risk]{1.20cm}{DPS / facilities}
      {Owns credited protection and hard interlock logic. SC may display it, but does not become the safety owner.}
      {}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Protection Is A Separate Lane}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.31\textwidth}
    \DuneFixedCard[dune accent]{2.45cm}{Observe}
      {SC displays protection state, alarms on it, and archives it for operators.}
      {This is monitoring.}
  \end{column}
  \begin{column}{0.31\textwidth}
    \DuneFixedCard[dune decision]{2.45cm}{Request}
      {SC may request permitted actions through approved command paths.}
      {This is controlled operation.}
  \end{column}
  \begin{column}{0.31\textwidth}
    \DuneFixedCard[dune risk]{2.45cm}{Protect}
      {DPS, facilities, and approved interlocks own credited protection.}
      {This must survive ordinary software failures.}
  \end{column}
\end{columns}
\medskip
\DuneCard[dune risk]{Hard rule}
  {Do not credit a bridge, HMI, GitHub workflow, or normal service process as protection unless the designated protection owner explicitly approves that claim.}
  {}
\end{frame}
