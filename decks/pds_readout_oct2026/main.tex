\begin{frame}{Tests exposed a host-processing bottleneck}
  \textbf{Test: 16 channels, 120 kHz target, noise descriptors.}\par
  \medskip
  \renewcommand{\arraystretch}{1.3}
  \begin{tabularx}{\textwidth}{@{}Xrrr@{}}
    \toprule
    Host processing & NIC loss & Raw queue loss & Descriptor loss \\
    \midrule
    Callbacks on RX thread & 51.91\% & --- & --- \\
    Callbacks on workers & 0\% & 5.55\% & --- \\
    Separate descriptor workers & 0\% & 4.37\% & 0\% \\
    \bottomrule
  \end{tabularx}
  \bigskip
  Offload callbacks so RX keeps draining the NIC ring.\par
  \smallskip
  Separate descriptors so raw backlog does not block TPs.\par
  \medskip
  {\small\color{DuneTextMuted}Host software changes; NIC firmware unchanged.\par
    Separate short tests; descriptor populations vary. Not a losslessness qualification.}
\end{frame}
