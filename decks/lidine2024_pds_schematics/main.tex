\newcommand{\SchematicBackground}{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[white] (current page.south west) rectangle (current page.north east);
  \end{tikzpicture}%
}

\newcommand{\SchematicPlate}[5]{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north,draw=DuneInfo!50!DuneTitleSurface,
      fill=DuneSky!35!white,
      minimum width=8.00cm,minimum height=0.62cm,inner xsep=8pt,
      text=DuneInfo!50!DuneTitleSurface,font=\fontsize{16}{18}\selectfont]
      at ($(current page.north)+(0,-0.20)$) {#1};
    \node[anchor=north,draw=DuneInfo!50!DuneTitleSurface,
      fill=DuneCoral!60!white,
      minimum width=10.40cm,minimum height=0.42cm,inner xsep=7pt,
      text=DuneInfo!50!DuneTitleSurface,font=\fontsize{8.2}{9.4}\selectfont]
      at ($(current page.north)+(0,-1.03)$) {#3};
    \node[anchor=center,inner sep=0pt]
      at ($(current page.center)+(0,-0.10)$) {\input{#2}};
    \draw[DunePrimary,line width=1.15pt]
      ($(current page.south west)+(0.72,0.54)$) --
      ($(current page.south east)+(-0.72,0.54)$);
    \node[anchor=south west,inner sep=0pt,text=black!65,
      font=\fontsize{5.8}{6.8}\selectfont]
      at ($(current page.south west)+(0.72,0.16)$)
      {\href{#5}{Source: #4}\enspace\textbullet\enspace
       vector reconstruction; connectivity transcribed, not ERC verified};
    \node[anchor=south east,inner sep=0pt,text=black!65,
      font=\fontsize{6.2}{7}\selectfont]
      at ($(current page.south east)+(-0.72,0.16)$) {\insertframenumber};
  \end{tikzpicture}%
}
