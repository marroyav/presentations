% Deck-specific sizes built on the shared framework's TikZ styles and colours.
\tikzset{
  dps node/.style={
    dune repo muted,draw=none,fill=DunePaper,text=DuneOnAccent,
    rounded corners=0pt,font=\small,text width=2.8cm,
    minimum height=0.95cm,inner sep=5pt
  },
  dps controller/.style={dps node,fill=DuneSky},
  dps load/.style={dps node,fill=DuneCoral},
  dps flow/.style={
    dune flow,-{Stealth[length=1.7mm,width=1.3mm]},
    draw=DuneSky,line width=1pt
  },
  dps both/.style={
    dps flow,arrows={Stealth[length=1.7mm,width=1.3mm]-Stealth[length=1.7mm,width=1.3mm]}
  },
  dps power/.style={dps flow,draw=DuneCoral},
  dps label/.style={
    dune label,fill=none,text=DuneText,font=\small,inner sep=2pt
  },
  dps boundary/.style={draw=DuneBorder,fill=DuneSurface,line width=0.7pt},
  dps heading/.style={text=DuneText,font=\small\bfseries,inner sep=0pt}
}
