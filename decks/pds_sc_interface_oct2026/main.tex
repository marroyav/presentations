\begin{frame}{Define each SC point before using it}
  \begin{enumerate}
    \setlength{\itemsep}{0.8em}
    \item \textbf{Source:} device, channel, NodeId, units and scaling.
    \item \textbf{Access:} Control or Publish, with an owner.
    \item \textbf{Limits:} alarm levels, hysteresis and on-delay.
    \item \textbf{Handling:} scan/history rate, stale data and response.
  \end{enumerate}
  \medskip
  Unknown values stay blank. Protection actions need agreement with DPS.
\end{frame}
